$$

\begin{multline}
\frac{\partial R}{\partial t}
+\frac{\partial u_j R}{\partial x_j}

\\
C_1 R S + f_1 C_{2kw}\frac{R}{S}\frac{\partial R}{\partial x_j}\frac{\partial S}{\partial x_j}
-(1-f_1)\min\left(C_{2kw}R^2\frac{\frac{\partial S}{\partial x_j}\frac{\partial S}{\partial x_j}}{S^2},C_m\frac{\partial R}{\partial x_j}\frac{\partial R}{\partial x_j}\right)
\end{multline}
\end{split}
$$
